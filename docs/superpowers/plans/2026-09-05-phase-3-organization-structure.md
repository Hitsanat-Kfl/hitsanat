**Goal:** Implement Member API (Stage 1 & 2) and Family API endpoints for the Organization Structure module.

**Architecture:** DDD layered architecture with Express routes → Use Cases → Repository → Drizzle ORM. Each module follows `domain/`, `application/`, `infrastructure/`, `presentation/` structure. Business rules enforced at domain and validation layers.

**Tech Stack:** Express, Drizzle ORM, PostgreSQL, Zod validation, Vitest, Biome

## Global Constraints

- Node >=20, pnpm 9.15.9, TypeScript strict
- Biome for formatting/linting (no ESLint)
- All dates stored as Gregorian/UTC in PostgreSQL
- Phone numbers validated with `phoneSchema` from `@repo/validation`
- UUIDs generated by PostgreSQL `defaultRandom()`
- All API responses follow `{ success, data?, error? }` shape from `@repo/validation`
- Module structure: `domain/` (entities, errors, repositories), `application/` (dto, use-cases), `infrastructure/` (repositories), `presentation/` (controllers, routes, validators)
- Use `getDb()` from `@repo/database` for DB access (lazy singleton)
- Auth middleware: `requireAuth()` and `requireScopePermission()` from `@repo/auth/middleware`

```typescript
import { eq, ilike, or, desc, asc, count, type SQL } from "drizzle-orm";
import { getDb } from "@repo/database";
import { members, type Member, type NewMember } from "@repo/database/schema";
import type {
  MemberRepository,
} from "../../domain/repositories/member.repository.js";
import type { PaginationParams, PaginatedResponse } from "@repo/schemas";

export class DrizzleMemberRepository implements MemberRepository {
  private db = getDb();

  async findById(id: string): Promise<Member | null> {
    const result = await this.db.query.members.findFirst({
      where: eq(members.id, id),
    });
    return result ?? null;
  }

  async findByPhoneNumber(phoneNumber: string): Promise<Member | null> {
    const result = await this.db.query.members.findFirst({
      where: eq(members.phoneNumber, phoneNumber),
    });
    return result ?? null;
  }

  async findMany(
    params: PaginationParams
  ): Promise<PaginatedResponse<Member>> {
    const page = params.page ?? 1;
    const limit = params.limit ?? 20;
    const offset = (page - 1) * limit;

    const conditions: SQL[] = [];

    if (params.search) {
      conditions.push(
        or(
          ilike(members.fullName, `%${params.search}%`),
          ilike(members.christianName, `%${params.search}%`),
          ilike(members.phoneNumber, `%${params.search}%`)
        )!
      );
    }

    const where = conditions.length > 0 ? conditions[0] : undefined;

    const [data, totalResult] = await Promise.all([
      this.db.query.members.findMany({
        where,
        orderBy: [desc(members.createdAt)],
        limit,
        offset,
      }),
      this.db.select({ value: count() }).from(members).where(where),
    ]);

    const total = totalResult[0]?.value ?? 0;

    return {
      data,
      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit),
      },
    };
  }

  async create(data: NewMember): Promise<Member> {
    const result = await this.db
      .insert(members)
      .values(data)
      .returning();
    return result[0];
  }

  async update(id: string, data: Partial<NewMember>): Promise<Member> {
    const result = await this.db
      .update(members)
      .set({ ...data, updatedAt: new Date() })
      .where(eq(members.id, id))
      .returning();
    if (result.length === 0) throw new Error(`Member not found: ${id}`);
    return result[0];
  }

  async delete(id: string): Promise<void> {
    await this.db.delete(members).where(eq(members.id, id));
  }

  async findBySubDepartment(subDepartmentId: string): Promise<Member[]> {
    const { subDepartmentMembers } = await import("@repo/database/schema");
    const results = await this.db.query.subDepartmentMembers.findMany({
      where: eq(subDepartmentMembers.subDepartmentId, subDepartmentId),
      with: { member: true },
    });
    return results.map((r) => r.member);
  }

  async findByFamily(familyId: string): Promise<Member[]> {
    const { familyMembers } = await import("@repo/database/schema");
    const results = await this.db.query.familyMembers.findMany({
      where: eq(familyMembers.familyId, familyId),
      with: { member: true },
    });
    return results.map((r) => r.member);
  }

  async findActive(): Promise<Member[]> {
    return this.db.query.members.findMany({
      where: eq(members.isActive, true),
    });
  }

  async countActive(): Promise<number> {
    const result = await this.db
      .select({ value: count() })
      .from(members)
      .where(eq(members.isActive, true));
    return result[0]?.value ?? 0;
  }
}
```

```typescript
import { getDb } from "@repo/database";
import {
  subDepartmentMembers,
  familyMembers,
  type Member,
} from "@repo/database/schema";
import type { MemberRepository } from "../../domain/repositories/member.repository.js";
import {
  MemberNotFoundError,
  MemberAlreadyInSubDepartmentError,
} from "../../domain/errors/member.error.js";

/**
 * Update Member Stage 2 Use Case
 * Enriches member profile with sub-department assignments, family link, photo, telegram
 */
export class UpdateMemberStage2UseCase {
  constructor(private readonly memberRepository: MemberRepository) {}

  async execute(
    memberId: string,
    data: {
      subDepartmentIds: string[];
      familyId?: string;
      photoUrl?: string;
      telegramUsername?: string;
    }
  ): Promise<Member> {
    const member = await this.memberRepository.findById(memberId);
    if (!member) {
      throw new MemberNotFoundError(memberId);
    }

    const db = getDb();

    // Update member profile fields
    await this.memberRepository.update(memberId, {
      photoUrl: data.photoUrl ?? member.photoUrl,
      telegramUsername: data.telegramUsername ?? member.telegramUsername,
    });

    // Assign sub-departments (create join records)
    for (const subDeptId of data.subDepartmentIds) {
      // Check if already assigned
      const existing = await db.query.subDepartmentMembers.findFirst({
        where: (sdm: any, { and }: any) =>
          and(
            sdm.memberId === memberId,
            sdm.subDepartmentId === subDeptId
          ),
      });

      if (existing) {
        throw new MemberAlreadyInSubDepartmentError(memberId, subDeptId);
      }

      await db.insert(subDepartmentMembers).values({
        memberId,
        subDepartmentId: subDeptId,
        role: "MEMBER",
      });
    }

    // Assign family if provided
    if (data.familyId) {
      // Remove from old family if exists
      const oldFamilyLink = await db.query.familyMembers.findFirst({
        where: (fm: any, { eq }: any) => eq(fm.memberId, memberId),
      });
      if (oldFamilyLink) {
        await db
          .delete(familyMembers)
          .where((fm: any, { eq }: any) => eq(fm.id, oldFamilyLink.id));
      }

      await db.insert(familyMembers).values({
        familyId: data.familyId,
        memberId,
      });
    }

    // Return updated member
    const updated = await this.memberRepository.findById(memberId);
    return updated!;
  }
}
```

```typescript
import type { Request, Response } from "express";
import { CreateMemberStage1UseCase } from "../../application/use-cases/create-member-stage1.use-case.js";
import { UpdateMemberStage2UseCase } from "../../application/use-cases/update-member-stage2.use-case.js";
import { GetMembersUseCase } from "../../application/use-cases/get-members.use-case.js";
import { GetMemberByIdUseCase } from "../../application/use-cases/get-member-by-id.use-case.js";
import { DrizzleMemberRepository } from "../../infrastructure/repositories/member.repository.js";
import { MemberAlreadyExistsError, MemberNotFoundError } from "../../domain/errors/member.error.js";

const memberRepository = new DrizzleMemberRepository();
const createStage1 = new CreateMemberStage1UseCase(memberRepository);
const updateStage2 = new UpdateMemberStage2UseCase(memberRepository);
const getMembers = new GetMembersUseCase(memberRepository);
const getMemberById = new GetMemberByIdUseCase(memberRepository);

export class MemberController {
  static async list(req: Request, res: Response) {
    try {
      const { search, page, limit } = req.query;
      const result = await getMembers.execute({
        search: search as string,
        page: page ? Number.parseInt(page as string, 10) : undefined,
        limit: limit ? Number.parseInt(limit as string, 10) : undefined,
      });
      res.json({ success: true, data: result.data, pagination: result.pagination });
    } catch (error) {
      res.status(500).json({ success: false, error: "Internal Server Error" });
    }
  }

  static async getById(req: Request, res: Response) {
    try {
      const member = await getMemberById.execute(req.params.id);
      res.json({ success: true, data: member });
    } catch (error) {
      if (error instanceof MemberNotFoundError) {
        return res.status(404).json({ success: false, error: error.message });
      }
      res.status(500).json({ success: false, error: "Internal Server Error" });
    }
  }

  static async createStage1(req: Request, res: Response) {
    try {
      const member = await createStage1.execute(req.body);
      res.status(201).json({ success: true, data: member });
    } catch (error) {
      if (error instanceof MemberAlreadyExistsError) {
        return res.status(409).json({ success: false, error: error.message });
      }
      res.status(500).json({ success: false, error: "Internal Server Error" });
    }
  }

  static async updateStage2(req: Request, res: Response) {
    try {
      const member = await updateStage2.execute(req.params.id, req.body);
      res.json({ success: true, data: member });
    } catch (error) {
      if (error instanceof MemberNotFoundError) {
        return res.status(404).json({ success: false, error: error.message });
      }
      res.status(500).json({ success: false, error: "Internal Server Error" });
    }
  }
}
```

```typescript
// apps/api/src/modules/family/infrastructure/repositories/family.repository.ts

import { eq, and } from "drizzle-orm";
import { getDb } from "@repo/database";
import {
  families,
  familyMembers,
  members,
  type Family,
  type FamilyMember,
  type Member,
} from "@repo/database/schema";
import type { FamilyRepository } from "../../domain/repositories/family.repository.js";

export class DrizzleFamilyRepository implements FamilyRepository {
  private db = getDb();

  async findById(id: string): Promise<Family | null> {
    const result = await this.db.query.families.findFirst({
      where: eq(families.id, id),
    });
    return result ?? null;
  }

  async findByName(
    familyName: string,
    academicYear: string
  ): Promise<Family | null> {
    const result = await this.db.query.families.findFirst({
      where: and(
        eq(families.familyName, familyName),
        eq(families.academicYear, academicYear)
      ),
    });
    return result ?? null;
  }

  async create(data: {
    familyName: string;
    academicYear: string;
  }): Promise<Family> {
    const result = await this.db.insert(families).values(data).returning();
    return result[0];
  }

  async updateFather(familyId: string, memberId: string | null): Promise<void> {
    await this.db
      .update(families)
      .set({ fatherMemberId: memberId })
      .where(eq(families.id, familyId));
  }

  async updateMother(familyId: string, memberId: string | null): Promise<void> {
    await this.db
      .update(families)
      .set({ motherMemberId: memberId })
      .where(eq(families.id, familyId));
  }

  async addMember(familyId: string, memberId: string): Promise<FamilyMember> {
    const result = await this.db
      .insert(familyMembers)
      .values({ familyId, memberId })
      .returning();
    return result[0];
  }

  async removeMember(familyId: string, memberId: string): Promise<void> {
    await this.db
      .delete(familyMembers)
      .where(
        and(
          eq(familyMembers.familyId, familyId),
          eq(familyMembers.memberId, memberId)
        )
      );
  }

  async getMembers(familyId: string): Promise<Member[]> {
    const links = await this.db.query.familyMembers.findMany({
      where: eq(familyMembers.familyId, familyId),
      with: { member: true },
    });
    return links.map((l) => l.member);
  }

  async listByAcademicYear(academicYear: string): Promise<Family[]> {
    return this.db.query.families.findMany({
      where: eq(families.academicYear, academicYear),
    });
  }

  async isMemberInFamily(
    memberId: string,
    familyId: string
  ): Promise<boolean> {
    const result = await this.db.query.familyMembers.findFirst({
      where: and(
        eq(familyMembers.memberId, memberId),
        eq(familyMembers.familyId, familyId)
      ),
    });
    return result !== undefined;
  }
}
```

```typescript
// apps/api/src/modules/family/application/use-cases/assign-parent.use-case.ts

import type { Family } from "@repo/database/schema";
import type { FamilyRepository } from "../../domain/repositories/family.repository.js";
import type { MemberRepository } from "../../../member/domain/repositories/member.repository.js";
import {
  FamilyNotFoundError,
  FatherAlreadyAssignedError,
  MotherAlreadyAssignedError,
  MemberNotFoundError,
  MemberNotEligibleForParentRoleError,
} from "../../domain/errors/family.error.js";

export class AssignParentUseCase {
  constructor(
    private readonly familyRepository: FamilyRepository,
    private readonly memberRepository: MemberRepository
  ) {}

  async execute(
    familyId: string,
    data: { memberId: string; role: "father" | "mother" }
  ): Promise<Family> {
    const family = await this.familyRepository.findById(familyId);
    if (!family) throw new FamilyNotFoundError(familyId);

    const member = await this.memberRepository.findById(data.memberId);
    if (!member) throw new MemberNotFoundError(data.memberId);

    // BR-004: Check gender eligibility
    if (data.role === "father" && member.gender !== "Male") {
      throw new MemberNotEligibleForParentRoleError(data.memberId, "father");
    }
    if (data.role === "mother" && member.gender !== "Female") {
      throw new MemberNotEligibleForParentRoleError(data.memberId, "mother");
    }

    // BR-004: Max one father, max one mother
    if (data.role === "father" && family.fatherMemberId) {
      throw new FatherAlreadyAssignedError(familyId);
    }
    if (data.role === "mother" && family.motherMemberId) {
      throw new MotherAlreadyAssignedError(familyId);
    }

    if (data.role === "father") {
      await this.familyRepository.updateFather(familyId, data.memberId);
    } else {
      await this.familyRepository.updateMother(familyId, data.memberId);
    }

    // Also add to family_members if not already there
    const alreadyMember = await this.familyRepository.isMemberInFamily(
      data.memberId,
      familyId
    );
    if (!alreadyMember) {
      await this.familyRepository.addMember(familyId, data.memberId);
    }

    return (await this.familyRepository.findById(familyId))!;
  }
}
```

```typescript
// apps/api/src/modules/family/presentation/controllers/family.controller.ts

import type { Request, Response } from "express";
import { DrizzleFamilyRepository } from "../../infrastructure/repositories/family.repository.js";
import { DrizzleMemberRepository } from "../../../member/infrastructure/repositories/member.repository.js";
import { CreateFamilyUseCase } from "../../application/use-cases/create-family.use-case.js";
import { AssignParentUseCase } from "../../application/use-cases/assign-parent.use-case.js";
import { AddMemberToFamilyUseCase } from "../../application/use-cases/add-member-to-family.use-case.js";
import { GetFamilyRosterUseCase } from "../../application/use-cases/get-family-roster.use-case.js";
import {
  FamilyNotFoundError,
  FamilyAlreadyExistsError,
  FatherAlreadyAssignedError,
  MotherAlreadyAssignedError,
  MemberNotEligibleForParentRoleError,
  MemberAlreadyInFamilyError,
} from "../../domain/errors/family.error.js";

const familyRepository = new DrizzleFamilyRepository();
const memberRepository = new DrizzleMemberRepository();
const createFamily = new CreateFamilyUseCase(familyRepository);
const assignParent = new AssignParentUseCase(familyRepository, memberRepository);
const addMember = new AddMemberToFamilyUseCase(familyRepository, memberRepository);
const getRoster = new GetFamilyRosterUseCase(familyRepository);

export class FamilyController {
  static async create(req: Request, res: Response) {
    try {
      const family = await createFamily.execute(req.body);
      res.status(201).json({ success: true, data: family });
    } catch (error) {
      if (error instanceof FamilyAlreadyExistsError) {
        return res.status(409).json({ success: false, error: error.message });
      }
      res.status(500).json({ success: false, error: "Internal Server Error" });
    }
  }

  static async assignParent(req: Request, res: Response) {
    try {
      const family = await assignParent.execute(req.params.id, req.body);
      res.json({ success: true, data: family });
    } catch (error) {
      if (error instanceof FamilyNotFoundError) {
        return res.status(404).json({ success: false, error: error.message });
      }
      if (error instanceof FatherAlreadyAssignedError || error instanceof MotherAlreadyAssignedError) {
        return res.status(409).json({ success: false, error: error.message });
      }
      if (error instanceof MemberNotEligibleForParentRoleError) {
        return res.status(422).json({ success: false, error: error.message });
      }
      res.status(500).json({ success: false, error: "Internal Server Error" });
    }
  }

  static async addMember(req: Request, res: Response) {
    try {
      const link = await addMember.execute(req.params.id, req.body.memberId);
      res.status(201).json({ success: true, data: link });
    } catch (error) {
      if (error instanceof FamilyNotFoundError || error instanceof MemberNotFoundError) {
        return res.status(404).json({ success: false, error: error.message });
      }
      if (error instanceof MemberAlreadyInFamilyError) {
        return res.status(409).json({ success: false, error: error.message });
      }
      res.status(500).json({ success: false, error: "Internal Server Error" });
    }
  }

  static async getRoster(req: Request, res: Response) {
    try {
      const roster = await getRoster.execute(req.params.id);
      res.json({ success: true, data: roster });
    } catch (error) {
      if (error instanceof FamilyNotFoundError) {
        return res.status(404).json({ success: false, error: error.message });
      }
      res.status(500).json({ success: false, error: "Internal Server Error" });
    }
  }
}
```
